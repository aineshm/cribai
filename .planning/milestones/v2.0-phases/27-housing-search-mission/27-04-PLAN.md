    ```typescript
    import type { MissionStep, StepContext, StepResult } from '../../executor';
    import type { ResearchedListing, ShortlistReport, ShortlistItem } from '@campusnest/types';

    interface RankedListing extends ResearchedListing {
      compositeScore: number
    }

    /**
     * Generate one-sentence reasoning for each listing using Gemini.
     * Falls back to template string if Gemini call fails.
     */
    async function generateReasonings(
      listings: RankedListing[],
      preferences: string | undefined,
    ): Promise<string[]> {
      try {
        const { GoogleGenAI } = await import('@google/genai');
        const ai = new GoogleGenAI({ apiKey: process.env.GEMINI_API_KEY ?? '' });

        const listingDescriptions = listings.map((l, i) =>
          `${i + 1}. ${l.address} — $${l.rentMonthly}/mo, ${l.bedrooms ?? '?'}BR, ` +
          `fairness: ${l.fairnessScore ?? 'n/a'}/10, ` +
          `reviews: ${l.reviewRating ?? 'n/a'}/5, ` +
          `walk score: ${l.walkScore ?? 'n/a'}`
        ).join('\n');

        const prefLine = preferences ? `Student preferences: "${preferences}"\n` : '';
        const prompt = `${prefLine}For each listing, write a single sentence explaining why it ranked where it did. Be specific about price, walkability, or reviews. Keep each under 20 words.

Listings (ranked #1 = best):
${listingDescriptions}

Return ONLY a JSON array of strings in order, e.g. ["Excellent walkability and fair price.", "Good reviews but slightly over budget."]`;

        const response = await ai.models.generateContent({
          model: 'gemini-2.5-flash',
          contents: prompt,
        });

        const text = response.text?.trim() ?? '[]';
        const reasonings = JSON.parse(text) as string[];
        return listings.map((_, i) => reasonings[i] ?? buildFallbackReasoning(listings[i]!));
      } catch {
        return listings.map(buildFallbackReasoning);
      }
    }

    function buildFallbackReasoning(l: RankedListing): string {
      const parts: string[] = [];
      if (l.fairnessScore != null && l.fairnessScore >= 7) parts.push('priced fairly');
      if (l.reviewRating != null && l.reviewRating >= 4) parts.push('highly rated');
      if (l.walkScore != null && l.walkScore >= 70) parts.push('walkable location');
      return parts.length > 0
        ? `Strong choice: ${parts.join(', ')}.`
        : 'Matches your search criteria.';
    }

    export const generateReportStep: MissionStep = {
      id: 'generate_report',
      label: 'Generating shortlist',
      run: async (ctx: StepContext): Promise<StepResult> => {
        const rankedListings = (ctx.state.rankedListings ?? []) as RankedListing[];
        const totalSearched = (ctx.state.totalSearched ?? 0) as number;
        const topListings = rankedListings.slice(0, 5);  // max 5 in report
        const preferences = (ctx.input as { preferences?: string }).preferences;

        const reasonings = await generateReasonings(topListings, preferences);

        const items: ShortlistItem[] = topListings.map((listing, i) => ({
          rank: i + 1,
          listingId: listing.id,
          address: listing.address,
          rentMonthly: listing.rentMonthly,
          compositeScore: listing.compositeScore,
          fairnessScore: listing.fairnessScore,
          reviewRating: listing.reviewRating,
          walkScore: listing.walkScore,
          preferenceScore: listing.preferenceScore,
          reasoning: reasonings[i] ?? buildFallbackReasoning(listing),
        }));

        const report: ShortlistReport = {
          missionId: ctx.missionId,
          generatedAt: new Date().toISOString(),
          totalSearched,
          items,
        };

        return {
          output: { report },
          done: true,
        };
      },
    };
    ```

    ```tsx
    'use client';

    import type { ShortlistReport, ShortlistItem } from '@campusnest/types';

    import { Badge } from '@/components/ui/badge';
    import { MapPin, Star, TrendingUp, Wind } from 'lucide-react';

    interface HousingSearchResultCardProps {
      readonly report: ShortlistReport;
    }

    function ShortlistItemRow({ item, index }: { item: ShortlistItem; index: number }) {
      return (
        <div className="flex gap-4 py-3 border-b last:border-0">
          <div className="flex-shrink-0">
            <Badge variant={index === 0 ? 'default' : 'secondary'} className="w-8 justify-center">
              #{item.rank}
            </Badge>
          </div>
          <div className="flex-1 min-w-0">
            <div className="flex items-start justify-between gap-2">
              <div className="flex items-center gap-1 text-sm font-medium truncate">
                <MapPin className="h-3.5 w-3.5 flex-shrink-0 text-muted-foreground" />
                <span className="truncate">{item.address}</span>
              </div>
              <span className="flex-shrink-0 text-sm font-semibold">
                ${item.rentMonthly.toLocaleString()}/mo
              </span>
            </div>
            <div className="flex gap-3 mt-1 text-xs text-muted-foreground">
              {item.fairnessScore != null && (
                <span className="flex items-center gap-1">
                  <TrendingUp className="h-3 w-3" />
                  Fair {item.fairnessScore}/10
                </span>
              )}
              {item.reviewRating != null && (
                <span className="flex items-center gap-1">
                  <Star className="h-3 w-3" />
                  {item.reviewRating.toFixed(1)}/5

              )}
              {item.walkScore != null && (
                <span className="flex items-center gap-1">
                  <Wind className="h-3 w-3" />

                </span>
              )}
            </div>
            <p className="mt-1 text-xs text-muted-foreground italic">{item.reasoning}</p>
          </div>
        </div>
      );
    }

    export function HousingSearchResultCard({ report }: HousingSearchResultCardProps) {
      return (
        <Card>
          <CardHeader className="pb-2">
            <CardTitle className="text-base">
              Housing Shortlist
              <span className="ml-2 text-xs font-normal text-muted-foreground">
                {report.items.length} of {report.totalSearched} searched
              </span>
            </CardTitle>

          <CardContent>

                {report.items.map((item, i) => (
                  <ShortlistItemRow key={item.listingId} item={item} index={i} />
                ))}
              </div>
            )}
          </CardContent>
        </Card>
      );
    }
    ```
