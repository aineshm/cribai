    ```typescript
    import type { SupabaseClient } from '@supabase/supabase-js';
    import type { MissionStep, StepContext, StepResult } from '../../executor';
    import type { HousingSearchInput } from '@campusnest/types';
    import { searchListings } from '../../tools/handlers/search-listings';  // adjust path as needed

    /**
     * Build a natural-language query string from the housing search input.
     * Exported for unit testing.
     */
    export function buildSearchQuery(input: HousingSearchInput): string {
      const parts: string[] = [];
      if (input.bedrooms != null) parts.push(`${input.bedrooms} bedroom`);
      if (input.maxRent != null) parts.push(`under $${input.maxRent}/month`);
      if (input.preferences) parts.push(input.preferences);
      if (input.dealbreakers?.length) {
        parts.push(`excluding: ${input.dealbreakers.join(', ')}`);
      }
      return parts.length > 0
        ? parts.join(', ')
        : 'affordable student housing near campus';
    }

    export const searchListingsStep: MissionStep = {
      id: 'search_listings',
      label: 'Searching listings',
      tool: 'search_listings',
      run: async (ctx: StepContext): Promise<StepResult> => {
        const input = ctx.input as HousingSearchInput;
        const query = buildSearchQuery(input);

        const toolCtx = {
          supabase: ctx.supabase,
          campusId: ctx.campusId,
          campusSlug: ctx.campusSlug,
          userId: ctx.userId,
        };

        const toolArgs: Record<string, unknown> = {
          query,
          campus_slug: ctx.campusSlug,
          limit: 20,
        };
        if (input.bedrooms != null) {
          toolArgs.min_beds = input.bedrooms;
          toolArgs.max_beds = input.bedrooms;
        }
        if (input.maxRent != null) toolArgs.max_rent = input.maxRent;

        const result = await searchListings(toolArgs, toolCtx);

        // modelContext from search_listings is JSON — parse to get listings array.
        // Read the actual handler to confirm the exact shape; adjust parsing below.
        let rawListings: unknown[] = [];
        try {
          const parsed = JSON.parse(result.modelContext) as { listings?: unknown[] };
          rawListings = parsed.listings ?? [];
        } catch {
          rawListings = [];
        }

        return {
          output: { rawListings, totalSearched: rawListings.length },
        };
      },
    };
    ```

    ```typescript
    import type { MissionStep, StepContext, StepResult } from '../../executor';

    interface RawListing {
      id: string
      address: string
      rent_monthly?: number
      [key: string]: unknown
    }

    /**
     * Deduplicate listings by normalized address (lowercase, trimmed).
     * When duplicates exist, keep the one with the lowest rent.
     * Exported for unit testing.
     */
    export function deduplicateListings(listings: RawListing[]): RawListing[] {
      const seen = new Map<string, RawListing>();
      for (const listing of listings) {
        const key = listing.address.toLowerCase().trim();
        const existing = seen.get(key);
        if (!existing) {
          seen.set(key, listing);
        } else {
          const existingRent = existing.rent_monthly ?? Infinity;
          const newRent = listing.rent_monthly ?? Infinity;
          if (newRent < existingRent) seen.set(key, listing);
        }
      }
      return Array.from(seen.values());
    }

    /**
     * Clamp topN to the actual number of available listings.
     * Exported for unit testing.
     */
    export function clampTopN(listings: unknown[], topN: number): number {
      return Math.min(topN, listings.length);
    }

    export const deduplicateStep: MissionStep = {
      id: 'deduplicate',
      label: 'Filtering results',
      run: async (ctx: StepContext): Promise<StepResult> => {
        const rawListings = (ctx.state.rawListings ?? []) as RawListing[];
        const inputTopN = (ctx.input as { topN?: number }).topN ?? 5;

        const listings = deduplicateListings(rawListings);
        const topN = clampTopN(listings, inputTopN);

        return {
          output: { listings, topN },
        };
      },
    };
    ```
