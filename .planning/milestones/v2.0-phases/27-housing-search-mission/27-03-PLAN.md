Implement steps 3 and 4 of the Housing Search Mission: step 3 researches each shortlisted
listing by calling existing tool handlers (reviews, neighborhood, detail), step 4 scores and
ranks all researched listings using a weighted composite formula.

Purpose: Steps 3-4 are the analytical core of the mission. Step 3 gathers the data needed to
differentiate listings beyond price. Step 4 applies the scoring model (30/25/20/25 weights)
and sorts by composite score so step 5 can format the best results. The scoring math is pure
and fully unit testable without Supabase.

Output: researchListingsStep, rankAndScoreStep, and unit tests for the scoring functions.

    function extractReviewData(modelContext: string): { rating: number | null; snippet: string | null } {
      try {
        const parsed = JSON.parse(modelContext) as { rating?: number; reviews?: Array<{ text?: string }> };
        return {
          rating: parsed.rating ?? null,
          snippet: parsed.reviews?.[0]?.text?.slice(0, 200) ?? null,
        };
      } catch {
        return { rating: null, snippet: null };
      }
    }

    /**
     * Extract walk score from the get_neighborhood_info tool modelContext.
     * Shape depends on actual handler — adjust after reading the file.
     */
    function extractWalkScore(modelContext: string): number | null {
      try {
        const parsed = JSON.parse(modelContext) as { walkScore?: number; walk_score?: number };
        return parsed.walkScore ?? parsed.walk_score ?? null;
      } catch {
        return null;
      }
    }

    export const researchListingsStep: MissionStep = {
      id: 'research_listings',
      label: 'Researching top listings',
      run: async (ctx: StepContext): Promise<StepResult> => {
        const listings = (ctx.state.listings ?? []) as RawListing[];
        const topN = (ctx.state.topN ?? 5) as number;

    ```typescript
    import type { MissionStep, StepContext, StepResult } from '../../executor';
    import type { ResearchedListing, ShortlistItem } from '@campusnest/types';

    type ScoreDimension = 'fairness' | 'reviews' | 'walkability' | 'preference';

    /**
     * Normalize a raw score value to 0-1 range for a given dimension.
     * Returns 0.5 (neutral) when value is null.
     * Exported for unit testing.
     */
    export function normalizeScore(dim: ScoreDimension, value: number | null): number {
      if (value === null) return 0.5;
      switch (dim) {
        case 'fairness':    return Math.min(1, Math.max(0, (value - 1) / 9));
        case 'reviews':     return Math.min(1, Math.max(0, (value - 1) / 4));
        case 'walkability': return Math.min(1, Math.max(0, value / 100));
        case 'preference':  return Math.min(1, Math.max(0, value / 10));
      }
    }

    /**
     * Compute weighted composite score for a listing.
     * Weights: fairness 30%, reviews 25%, walkability 20%, preference 25%.
     * Exported for unit testing.
     */
    export function scoreComposite(scores: {
      fairness: number | null;
      reviews: number | null;
      walkability: number | null;
      preference: number | null;
    }): number {
      return (
        0.30 * normalizeScore('fairness', scores.fairness) +
        0.25 * normalizeScore('reviews', scores.reviews) +
        0.20 * normalizeScore('walkability', scores.walkability) +
        0.25 * normalizeScore('preference', scores.preference)
      );
    }

    /**
     * Ask Gemini to rate each listing's match to a preferences string (0-10 per listing).
     * Returns an array of scores in the same order as listings.
     * Returns all-null if no preferences or if Gemini call fails.
     */
    async function scorePreferenceMatch(
      listings: ResearchedListing[],
      preferences: string | undefined,
    ): Promise<Array<number | null>> {
      if (!preferences?.trim()) return listings.map(() => null);

      try {
        // Import Gemini client — read packages/ai/src/cribai.ts to find the exact import
        // path for the GoogleGenAI client used in this package. Use the same model.
        const { GoogleGenAI } = await import('@google/genai');
        const ai = new GoogleGenAI({ apiKey: process.env.GEMINI_API_KEY ?? '' });

        const listingDescriptions = listings
          .map((l, i) => `${i + 1}. ${l.address} — $${l.rentMonthly}/mo, ${l.bedrooms ?? '?'}BR, amenities: ${l.amenities.join(', ')}`)
          .join('\n');

        const prompt = `Rate how well each listing matches the student's preferences on a scale of 0-10.
Student preferences: "${preferences}"

Listings:
${listingDescriptions}

Return ONLY a JSON array of numbers in the same order as the listings, e.g. [7, 4, 9].
No explanation.`;

        const response = await ai.models.generateContent({
          model: 'gemini-2.5-flash',
          contents: prompt,
        });

        const text = response.text?.trim() ?? '[]';
        const scores = JSON.parse(text) as number[];
        return listings.map((_, i) => scores[i] ?? null);
      } catch {
        return listings.map(() => null);
      }
    }

    /**
     * Score and rank a list of researched listings by composite score.
     * Exported for unit testing (pass preferenceScores directly to avoid Gemini call).
     */
    export function rankAndScore(
      listings: ResearchedListing[],
      preferenceScores: Array<number | null>,
    ): Array<ResearchedListing & { compositeScore: number }> {
      return listings
        .map((listing, i) => ({
          ...listing,
          preferenceScore: preferenceScores[i] ?? null,
          compositeScore: scoreComposite({
            fairness: listing.fairnessScore,
            reviews: listing.reviewRating,
            walkability: listing.walkScore,
            preference: preferenceScores[i] ?? null,
          }),
        }))
        .sort((a, b) => b.compositeScore - a.compositeScore);
    }

    export const rankAndScoreStep: MissionStep = {
      id: 'rank_and_score',
      label: 'Ranking listings',
      run: async (ctx: StepContext): Promise<StepResult> => {
        const researchedListings = (ctx.state.researchedListings ?? []) as ResearchedListing[];
        const preferences = (ctx.input as { preferences?: string }).preferences;

        const preferenceScores = await scorePreferenceMatch(researchedListings, preferences);
        const rankedListings = rankAndScore(researchedListings, preferenceScores);

        return { output: { rankedListings } };
      },
    };
    ```
