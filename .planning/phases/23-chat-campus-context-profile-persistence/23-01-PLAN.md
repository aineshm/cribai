async function getDefaultCampusSlug(supabase: SupabaseClient): Promise<string> {
  const { data } = await supabase
    .from('campus_configs')
    .select('slug')
    .order('created_at', { ascending: true })
    .limit(1)
    .single();
  return data?.slug ?? 'uw-madison';
}
