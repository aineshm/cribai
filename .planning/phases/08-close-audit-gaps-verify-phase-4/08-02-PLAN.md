```typescript
import { isDevAuthEnabled, getDevUserById, DEFAULT_DEV_USER, DEV_USER_COOKIE } from '../../../lib/dev-auth';
import { createServerComponentClient, createSecretClient } from '@campusnest/supabase/server';

async function resolveUserId(): Promise<{ userId: string | null; supabase: ReturnType<typeof createServerComponentClient> }> {
  const cookieStore = await cookies();
  const supabase = createServerComponentClient(cookieStore);
  if (isDevAuthEnabled()) {
    const selectedId = cookieStore.get(DEV_USER_COOKIE)?.value;
    const devUser = selectedId ? getDevUserById(selectedId) : DEFAULT_DEV_USER;
    return { userId: devUser?.id ?? DEFAULT_DEV_USER.id, supabase };
  }
  const { data: { user }, error } = await supabase.auth.getUser();
  return { userId: (!error && user) ? user.id : null, supabase };
}

// CRITICAL: use service-role client for DB writes in dev mode
const writeClient = isDevAuthEnabled() ? createSecretClient() : supabase;
```

**Step 1: Rebuild PageIndex**
```yaml
- name: Rebuild PageIndex for CribAI context
  if: success() && steps.embed.outcome == 'success'
  id: pageindex
  env:
    SUPABASE_URL: ${{ secrets.SUPABASE_URL }}
    SUPABASE_KEY: ${{ secrets.SUPABASE_SECRET_KEY }}
  run: |
    RESPONSE=$(curl -s -w "\n%{http_code}" -X POST \
      "${SUPABASE_URL}/functions/v1/rebuild-pageindex" \
      -H "Authorization: Bearer ${SUPABASE_KEY}" \
      -H "Content-Type: application/json")
    HTTP_CODE=$(echo "$RESPONSE" | tail -1)
    BODY=$(echo "$RESPONSE" | head -n -1)
    echo "HTTP $HTTP_CODE: $BODY"
    if [ "$HTTP_CODE" -ge 400 ]; then
      echo "::error::PageIndex rebuild failed with HTTP $HTTP_CODE"
      exit 1
    fi
```

**Step 2: Write PageIndex summary**
```yaml
- name: Write PageIndex rebuild summary
  if: always() && steps.pageindex.outcome != 'skipped'
  env:
    PAGEINDEX_OUTCOME: ${{ steps.pageindex.outcome }}
  run: |
    echo "" >> "$GITHUB_STEP_SUMMARY"
    echo "## PageIndex Rebuild" >> "$GITHUB_STEP_SUMMARY"
    echo "" >> "$GITHUB_STEP_SUMMARY"
    if [ "$PAGEINDEX_OUTCOME" = "failure" ]; then
      echo "> **FAILED** -- Check logs for details" >> "$GITHUB_STEP_SUMMARY"
    elif [ "$PAGEINDEX_OUTCOME" = "success" ]; then
      echo "> **SUCCESS**" >> "$GITHUB_STEP_SUMMARY"
    else
      echo "> **SKIPPED** (embedding step did not succeed)" >> "$GITHUB_STEP_SUMMARY"
    fi
```

The gate `if: success() && steps.embed.outcome == 'success'` ensures PageIndex only rebuilds when fresh embeddings were just generated. The pipeline order is: scrape -> fairness -> embeddings -> PageIndex rebuild.
